1环境与安装（统一用 uv）

# 安装 uv
curl -LsSf https://astral.sh/uv/install.sh | sh                          # macOS / Linux
powershell -ExecutionPolicy ByPass -c "irm https://astral.sh/uv/install.ps1 | iex"   # Windows

# 国内镜像（一次即可，装库全靠它提速）
export UV_DEFAULT_INDEX="https://pypi.tuna.tsinghua.edu.cn/simple"       # macOS / Linux
$env:UV_DEFAULT_INDEX="https://pypi.tuna.tsinghua.edu.cn/simple"         # Windows PowerShell

# 统一 Python 版本（全班一致，避免"我这能跑你那里报错"）
uv python pin --global 3.11          # 全局默认版本
uv python install 3.11               # 可选：提前装下来，避免首次使用时才下载

# 建项目 → 钉版本 → 加依赖 → 运行（不用手动激活环境）
mkdir robot_training
cd robot_training
uv init
uv python pin 3.11                   # 项目级钉死（比全局更优先）
uv add opencv-python numpy ultralytics
uv run python camera.py

# 拿到别人的项目（含 pyproject.toml + uv.lock）时复现同样的环境
uv sync

# 临时跑个脚本、不想建项目
uv run --with opencv-python python demo.py

# 装命令行工具（如标注工具 labelImg）
uv tool install labelImg

2读摄像头 / 存图

import cv2

cap = cv2.VideoCapture(0)            # 打不开就试 1、2
while True:
    ok, frame = cap.read()
    if not ok:
        break
    cv2.imshow("win", frame)
    k = cv2.waitKey(1) & 0xFF
    if k == ord("s"):
        cv2.imwrite("shot.jpg", frame)
    if k == ord("q"):
        break
cap.release()
cv2.destroyAllWindows()

3YOLO 推理

from ultralytics import YOLO

model = YOLO("yolo11n.pt")                    # 或自己的 best.pt
results = model(frame, conf=0.4, verbose=False)
r = results[0]

for box in r.boxes:
    cls_id = int(box.cls[0])
    conf   = float(box.conf[0])
    x1, y1, x2, y2 = box.xyxy[0].tolist()
    cx, cy = (x1 + x2) / 2, (y1 + y2) / 2     # 中心点，抓取常用它
    print(model.names[cls_id], conf, cx, cy)

annotated = r.plot()                          # 画好框的图

7像素坐标 → 机械臂坐标（9 点标定）

import cv2, numpy as np

# 至少 4 组对应点：图像里测到的像素点 → 机械臂走到的实际点（毫米）
pixel = np.float32([[120, 88], [430, 95], [437, 402], [115, 396]])
robot = np.float32([[100, 100], [300, 100], [300, 300], [100, 300]])

H, mask = cv2.findHomography(pixel, robot)          # 解出 3x3 变换矩阵

# 之后任意一个检测框中心点都能换算
pts = np.float32([[200, 150]]).reshape(-1, 1, 2)
xy = cv2.perspectiveTransform(pts, H).reshape(-1, 2)
print(xy)                                           # 机械臂该去的坐标
